    46→class DispatcherPass:
    47→    def __init__(self, root: Node, analyzer: Analyzer, source: str, stats):
    48→        self.root = root
    49→        self.analyzer = analyzer
    50→        self.source = source
    51→        self.stats = stats
    52→        self.max_states = 4096
    53→        self.max_paths = 96
    54→        self.max_work = 400000
    55→        self.emit_budget = 40000
    56→        self.work = 0
    57→        self.base_env = constant_environment(analyzer)
    58→
    59→    def apply(self, rounds: int = 8) -> int:
    60→        completed = 0
    61→        sys.setrecursionlimit(max(sys.getrecursionlimit(), 20000))
    62→        for _ in range(rounds):
    63→            changed = False
    64→            lists = self.statement_lists(self.root)
    65→            candidates: list[Dispatcher] = []
    66→            for statements in lists:
    67→                for index, statement in enumerate(statements):
    68→                    if statement.kind != "while":
    69→                        continue
    70→                    candidate = self.recognize(statements, index, statement)
    71→                    if candidate is not None:
    72→                        candidates.append(candidate)
    73→            candidates.sort(key=lambda item: item.loop.start, reverse=True)
    74→            for candidate in candidates:
    75→                try:
    76→                    replacement = self.recover(candidate)
    77→                except DispatcherBail:
    78→                    continue
    79→                except (RecursionError, OverflowError, ValueError, TypeError):
    80→                    continue
    81→                if replacement is None:
    82→                    continue
    83→                candidate.statements[candidate.index] = replacement
    84→                self.stats.dispatchers_removed += 1
    85→                completed += 1
    86→                changed = True
    87→            if not changed:
    88→                break
    89→        return completed

    91→    def statement_lists(self, root: Node) -> list[list[Node]]:
    92→        result: list[list[Node]] = []
    93→        seen: set[int] = set()
    94→
    95→        def add(statements: object) -> None:
    96→            if isinstance(statements, list) and id(statements) not in seen:
    97→                seen.add(id(statements))
    98→                result.append(statements)
    99→
   100→        add(root.fields.get("body", []))
   101→        for node in walk(root):
   102→            for key in ("body", "then", "else_"):
   103→                add(node.fields.get(key))
   104→            for _, body in node.fields.get("elifs", []):
   105→                add(body)
   106→        return result

   108→    def recognize(self, statements: list[Node], index: int, loop: Node) -> Dispatcher | None:
   109→        if not self.is_true(loop.get("cond")):
   110→            return None
   111→        body = loop.get("body", [])
   112→        if not body:
   113→            return None
   114→        head = body[0]
   115→        if head.kind != "assign" or len(head.get("targets", [])) != 1 or len(head.get("values", [])) != 1:
   116→            return None
   117→        target = head.get("targets", [])[0]
   118→        if target.kind != "name":
   119→            return None
   120→        state = binding_for(self.analyzer, target)
   121→        if state is None:
   122→            return None
   123→        transform = self.transition(head.get("values", [])[0], state)
   124→        if transform is None:
   125→            return None
   126→        initial = self.initial_value(statements, index, state)
   127→        if initial is None:
   128→            return None
   129→        tree = body[1:]
   130→        if len(tree) == 1 and tree[0].kind == "do":
   131→            tree = tree[0].get("body", [])
   132→        if not self.looks_like_dispatch(tree, state):
   133→            return None
   134→        return Dispatcher(loop, statements, index, state, initial, transform, tree)

   136→    def is_true(self, node: Node) -> bool:
   137→        return isinstance(node, Node) and node.kind == "bool" and node.get("value") is True
   138→
   139→    def transition(self, expression: Node, state: Binding) -> tuple[str, int | float] | None:
   140→        if expression.kind == "binop" and expression.get("op") == "-":
   141→            left = expression.get("left")
   142→            right = expression.get("right")
   143→            if self.is_state(right, state) and self.is_number_node(left):
   144→                return ("complement", left.get("value"))
   145→            if self.is_state(left, state) and self.is_number_node(right):
   146→                return ("negate", right.get("value"))
   147→        if expression.kind == "call":
   148→            name = self.dotted_name(expression.get("func"))
   149→            args = expression.get("args", [])
   150→            if name == "bit32.bxor" and len(args) == 2 and self.is_state(args[0], state) and self.is_number_node(args[1]):
   151→                return ("xor", args[1].get("value"))
   152→            if name == "bit32.bxor" and len(args) == 2 and self.is_state(args[1], state) and self.is_number_node(args[0]):
   153→                return ("xor", args[0].get("value"))
   154→        return None
   155→
   156→    def dotted_name(self, node: Node | None) -> str | None:
   157→        if not isinstance(node, Node):
   158→            return None
   159→        if node.kind == "name":
   160→            return node.get("name")
   161→        if node.kind == "indexname":
   162→            base = self.dotted_name(node.get("obj"))
   163→            return base + "." + node.get("name", "") if base else None
   164→        return None
   165→
   166→    def is_state(self, node: Node | None, state: Binding) -> bool:
   167→        return isinstance(node, Node) and node.kind == "name" and binding_for(self.analyzer, node) is state
   168→
   169→    def is_number_node(self, node: Node | None) -> bool:
   170→        return isinstance(node, Node) and node.kind == "number" and is_number(node.get("value"))
   171→
   172→    def initial_value(self, statements: list[Node], index: int, state: Binding) -> int | float | None:
   173→        for statement in reversed(statements[:index]):
   174→            if statement.kind == "local":
   175→                bindings = self.analyzer.declaration_bindings.get(id(statement), [])
   176→                names = statement.get("names", [])
   177→                for binding, value in zip(bindings, statement.get("values", [])):
   178→                    if binding is state and self.is_number_node(value):
   179→                        return value.get("value")
   180→                if state.name in names:
   181→                    return None
   182→            if statement.kind == "assign":
   183→                for target, value in zip(statement.get("targets", []), statement.get("values", [])):
   184→                    if target.kind == "name" and binding_for(self.analyzer, target) is state:
   185→                        return value.get("value") if self.is_number_node(value) else None
   186→            if any(child.kind == "name" and binding_for(self.analyzer, child) is state for child in self.node_children(statement)):
   187→                return None
   188→        return None
   189→
   190→    def node_children(self, node: Node) -> list[Node]:
   191→        result: list[Node] = []
   192→        for value in node.fields.values():
   193→            if isinstance(value, Node):
   194→                result.append(value)
   195→            elif isinstance(value, (list, tuple)):
   196→                for item in value:
   197→                    if isinstance(item, Node):
   198→                        result.append(item)
   199→                    elif isinstance(item, (list, tuple)):
   200→                        result.extend(item for item in item if isinstance(item, Node))
   201→        return result

   203→    def looks_like_dispatch(self, tree: list[Node], state: Binding) -> bool:
   204→        found = False
   205→        for node in tree:
   206→            for child in self.node_children(node):
   207→                if child.kind == "binop" and child.get("op") in {"<", "<=", ">", ">=", "==", "~="}:
   208→                    if self.is_state(child.get("left"), state) or self.is_state(child.get("right"), state):
   209→                        found = True
   210→        if found:
   211→            return True
   212→        return any(node.kind in {"break", "continue", "return"} for node in tree)
   213→
   214→    def apply_transform(self, value: int | float, transform: tuple[str, int | float]) -> int | float:
   215→        kind, constant = transform
   216→        if kind == "complement":
   217→            return constant - value
   218→        if kind == "negate":
   219→            return value - constant
   220→        return int(constant) ^ int(value)

   222→    def eval_data(self, node: Node | None, state: Binding, value: int | float, env: dict[int, Any]) -> Any:
   223→        if node is None:
   224→            return UNKNOWN
   225→        if self.is_state(node, state):
   226→            return value
   227→        if node.kind == "number" or node.kind == "string" or node.kind == "bool":
   228→            return node.get("value")
   229→        if node.kind == "nil":
   230→            return None
   231→        if node.kind == "name":
   232→            binding = binding_for(self.analyzer, node)
   233→            if binding is not None and binding.ident in env:
   234→                return env[binding.ident]
   235→            if binding is not None and binding.ident in self.base_env:
   236→                return self.base_env[binding.ident]
   237→            return UNKNOWN
   238→        if node.kind == "paren":
   239→            return self.eval_data(node.get("expr"), state, value, env)
   240→        if node.kind == "unop":
   241→            inner = self.eval_data(node.get("expr"), state, value, env)
   242→            if inner is UNKNOWN:
   243→                resolved = bool_const(node)
   244→                return resolved if resolved is not None else UNKNOWN
   245→            if node.get("op") == "not":
   246→                return not truthy(inner)
   247→            if node.get("op") == "-":
   248→                return -inner
   249→            if node.get("op") == "#" and isinstance(inner, bytes):
   250→                return len(inner)
   251→            return UNKNOWN
   252→        if node.kind == "binop":
   253→            operator = node.get("op")
   254→            if operator == "and":
   255→                left = self.eval_data(node.get("left"), state, value, env)
   256→                if left is UNKNOWN:
   257→                    resolved = bool_const(node)
   258→                    return resolved if resolved is not None else UNKNOWN
   259→                return self.eval_data(node.get("right"), state, value, env) if truthy(left) else left
   260→            if operator == "or":
   261→                left = self.eval_data(node.get("left"), state, value, env)
   262→                if left is UNKNOWN:
   263→                    resolved = bool_const(node)
   264→                    return resolved if resolved is not None else UNKNOWN
   265→                return left if truthy(left) else self.eval_data(node.get("right"), state, value, env)
   266→            left = self.eval_data(node.get("left"), state, value, env)
   267→            right = self.eval_data(node.get("right"), state, value, env)
   268→            if left is UNKNOWN or right is UNKNOWN:
   269→                return UNKNOWN
   270→            if operator == "..":
   271→                return left + right if isinstance(left, bytes) and isinstance(right, bytes) else UNKNOWN
   272→            if operator in {"==", "~=", "<", "<=", ">", ">="}:
   273→                try:
   274→                    if operator == "==":
   275→                        return left == right
   276→                    if operator == "~=":
   277→                        return left != right
   278→                    if operator == "<":
   279→                        return left < right
   280→                    if operator == "<=":
   281→                        return left <= right
   282→                    if operator == ">":
   283→                        return left > right
   284→                    return left >= right
   285→                except (TypeError, ValueError):
   286→                    return UNKNOWN
   287→            return numeric_result(operator, left, right)
   288→        if node.kind == "call":
   289→            name = self.dotted_name(node.get("func"))
   290→            args = [self.eval_data(argument, state, value, env) for argument in node.get("args", [])]
   291→            return builtin_call(name, args)
   292→        if node.kind == "ifexpr":
   293→            condition = self.eval_data(node.get("cond"), state, value, env)
   294→            if condition is UNKNOWN:
   295→                return UNKNOWN
   296→            if truthy(condition):
   297→                return self.eval_data(node.get("then"), state, value, env)
   298→            for branch_condition, branch_value in node.get("elifs", []):
   299→                branch = self.eval_data(branch_condition, state, value, env)
   300→                if branch is UNKNOWN:
   301→                    return UNKNOWN
   302→                if truthy(branch):
   303→                    return self.eval_data(branch_value, state, value, env)
   304→            return self.eval_data(node.get("else_"), state, value, env)
   305→        return UNKNOWN
   306→
   307→    def eval_state(self, node: Node, state: Binding, value: int | float, env: dict[int, Any]) -> int | float:
   308→        result = self.eval_data(node, state, value, env)
   309→        if result is UNKNOWN or not is_number(result):
   310→            raise DispatcherBail("dynamic state value")
   311→        return result
   312→
   313→    def eval_condition(self, node: Node, state: Binding, value: int | float, env: dict[int, Any]) -> bool:
   314→        result = self.eval_data(node, state, value, env)
   315→        if result is UNKNOWN:
   316→            raise DispatcherBail("dynamic dispatcher condition")
   317→        return truthy(result)

   319→    def substitute_state(self, node: Node, state: Binding, value: int | float) -> Node:
   320→        """Return a clone of node with every reference to the dispatcher
   321→        state replaced by its known numeric value at this program point."""
   322→        replacement = value_node(value)
   323→
   324→        def visit_pair(original: Node | None, clone: Node | None) -> None:
   325→            if original is None or clone is None:
   326→                return
   327→            if original.kind == "function":
   328→                raise StateSubstitutionError("state captured in payload closure")
   329→            if original.kind == "name" and binding_for(self.analyzer, original) is state:
   330→                clone.kind = replacement.kind
   331→                clone.fields = dict(replacement.fields)
   332→                return
   333→            for key, field in original.fields.items():
   334→                clone_field = clone.fields.get(key)
   335→                if isinstance(field, Node) and isinstance(clone_field, Node):
   336→                    visit_pair(field, clone_field)
   337→                elif isinstance(field, (list, tuple)) and isinstance(clone_field, (list, tuple)):
   338→                    visit_list(field, clone_field)
   339→
   340→        def visit_list(originals: Any, clones: Any) -> None:
   341→            for original, clone in zip(originals, clones):
   342→                if isinstance(original, Node) and isinstance(clone, Node):
   343→                    visit_pair(original, clone)
   344→                elif isinstance(original, (list, tuple)) and isinstance(clone, (list, tuple)):
   345→                    visit_list(original, clone)
   346→
   347→        clone = node.clone()
   348→        visit_pair(node, clone)
   349→        return clone

   351→    def execute(self, statements: list[Node], state: Binding, value: int | float, conditions: list[tuple[Node, bool]], actions: list[Node], depth: int = 0, env: dict[int, Any] | None = None) -> list[DispatchPath]:
   352→        self.work += 1
   353→        if self.work > self.max_work:
   354→            raise DispatcherBail("work limit")
   355→        if depth > 100:
   356→            raise DispatcherBail("dispatcher nesting limit")
   357→        paths: list[DispatchPath] = []
   358→        current_value = value
   359→        current_conditions = list(conditions)
   360→        current_actions = list(actions)
   361→        current_env = dict(env or {})
   362→        for statement_index, statement in enumerate(statements):
   363→            kind = statement.kind
   364→            if kind == "assign":
   365→                targets = statement.get("targets", [])
   366→                values = statement.get("values", [])
   367→                state_targets = [target for target in targets if target.kind == "name" and binding_for(self.analyzer, target) is state]
   368→                if state_targets:
   369→                    if len(targets) != 1 or len(values) != 1:
   370→                        raise DispatcherBail("state assignment mixed with other targets")
   371→                    try:
   372→                        current_value = self.eval_state(values[0], state, current_value, current_env)
   373→                    except DispatcherBail:
   374→                        if values[0].kind == "ifexpr":
   375→                            return self.expand_state_ifexpr(values[0], statements, statement_index, state, current_value, current_conditions, current_actions, current_env, depth)
   376→                        raise
   377→                    continue
   378→                if self.contains_state(statement, state):
   379→                    try:
   380→                        current_actions.append(self.substitute_state(statement, state, current_value))
   381→                    except StateSubstitutionError as exc:
   382→                        raise DispatcherBail(str(exc))
   383→                    if len(targets) == 1 and len(values) == 1 and targets[0].kind == "name":
   384→                        target_binding = binding_for(self.analyzer, targets[0])
   385→                        if target_binding is not None:
   386→                            assigned = self.eval_data(values[0], state, current_value, current_env)
   387→                            if assigned is UNKNOWN:
   388→                                current_env.pop(target_binding.ident, None)
   389→                            else:
   390→                                current_env[target_binding.ident] = assigned
   391→                    continue
   392→                if len(targets) == 1 and len(values) == 1 and targets[0].kind == "name":
   393→                    target_binding = binding_for(self.analyzer, targets[0])
   394→                    if target_binding is not None:
   395→                        assigned = self.eval_data(values[0], state, current_value, current_env)
   396→                        if assigned is UNKNOWN:
   397→                            current_env.pop(target_binding.ident, None)
   398→                        else:
   399→                            current_env[target_binding.ident] = assigned
   400→                current_actions.append(statement)
   401→                continue
   402→            if kind == "local":
   403→                if any(name in statement.get("names", []) for name in [state.name]) or self.contains_state(statement, state):
   404→                    if self.contains_state(statement, state):
   405→                        try:
   406→                            current_actions.append(self.substitute_state(statement, state, current_value))
   407→                        except StateSubstitutionError as exc:
   408→                            raise DispatcherBail(str(exc))
   409→                        for local_binding, local_value in zip(self.analyzer.declaration_bindings.get(id(statement), []), statement.get("values", [])):
   410→                            assigned = self.eval_data(local_value, state, current_value, current_env)
   411→                            if assigned is UNKNOWN:
   412→                                current_env.pop(local_binding.ident, None)
   413→                            else:
   414→                                current_env[local_binding.ident] = assigned
   415→                        continue
   416→                    raise DispatcherBail("state in local declaration")
   417→                for local_binding, local_value in zip(self.analyzer.declaration_bindings.get(id(statement), []), statement.get("values", [])):
   418→                    assigned = self.eval_data(local_value, state, current_value, current_env)
   419→                    if assigned is UNKNOWN:
   420→                        current_env.pop(local_binding.ident, None)
   421→                    else:
   422→                        current_env[local_binding.ident] = assigned
   423→                current_actions.append(statement)
   424→                continue
   425→            if kind == "if":
   426→                try:
   427→                    condition_value = self.eval_condition(statement.get("cond"), state, current_value, current_env)
   428→                except DispatcherBail:
   429→                    if self.contains_state(statement.get("cond"), state):
   430→                        raise
   431→                    return self.fork_if(statement, state, current_value, current_conditions, current_actions, depth, current_env)
   432→                if condition_value:
   433→                    selected = statement.get("then", [])
   434→                else:
   435→                    selected = self.else_chain(statement)
   436→                paths.extend(self.execute(selected, state, current_value, current_conditions, current_actions, depth + 1, current_env))
   437→                return paths
   438→            if kind == "do":
   439→                paths.extend(self.execute(statement.get("body", []), state, current_value, current_conditions, current_actions, depth + 1, current_env))
   440→                return paths
   441→            if kind in {"while", "repeat", "fornum", "forin"}:
   442→                if self.contains_state(statement, state):
   443→                    try:
   444→                        current_actions.append(self.substitute_state(statement, state, current_value))
   445→                    except StateSubstitutionError as exc:
   446→                        raise DispatcherBail(str(exc))
   447→                    continue
   448→                current_actions.append(statement)
   449→                continue
   450→            if kind in {"call", "methodcall"}:
   451→                if self.contains_state(statement, state):
   452→                    try:
   453→                        current_actions.append(self.substitute_state(statement, state, current_value))
   454→                    except StateSubstitutionError as exc:
   455→                        raise DispatcherBail(str(exc))
   456→                    continue
   457→                current_actions.append(statement)
   458→                continue
   459→            if kind == "return":
   460→                if any(self.contains_state(value, state) for value in statement.get("values", [])):
   461→                    substituted = []
   462→                    try:
   463→                        for value_node_item in statement.get("values", []):
   464→                            substituted.append(self.substitute_state(value_node_item, state, current_value))
   465→                    except StateSubstitutionError as exc:
   466→                        raise DispatcherBail(str(exc))
   467→                    paths.append(DispatchPath(current_conditions, current_actions, ("return", substituted), dict(current_env)))
   468→                    return paths
   469→                paths.append(DispatchPath(current_conditions, current_actions, ("return", statement.get("values", [])), dict(current_env)))
   470→                return paths
   471→            if kind == "break":
   472→                paths.append(DispatchPath(current_conditions, current_actions, ("break",), dict(current_env)))
   473→                return paths
   474→            if kind == "continue":
   475→                paths.append(DispatchPath(current_conditions, current_actions, ("edge", current_value), dict(current_env)))
   476→                return paths
   477→            raise DispatcherBail("unsupported dispatcher statement " + kind)
   478→        paths.append(DispatchPath(current_conditions, current_actions, ("edge", current_value), dict(current_env)))
   479→        return paths

   481→    def expand_state_ifexpr(self, expression: Node, statements: list[Node], index: int, state: Binding, value: int | float, conditions: list[tuple[Node, bool]], actions: list[Node], env: dict[int, Any], depth: int) -> list[DispatchPath]:
   482→        branches: list[tuple[Node | None, Node, list[tuple[Node, bool]]]] = []
   483→        prior: list[tuple[Node, bool]] = []
   484→        first_condition = expression.get("cond")
   485→        branches.append((first_condition, expression.get("then"), prior + [(first_condition, True)]))
   486→        prior = prior + [(first_condition, False)]
   487→        for branch_condition, branch_value in expression.get("elifs", []):
   488→            branches.append((branch_condition, branch_value, prior + [(branch_condition, True)]))
   489→            prior = prior + [(branch_condition, False)]
   490→        branches.append((None, expression.get("else_"), list(prior)))
   491→        result: list[DispatchPath] = []
   492→        for condition, leaf, branch_conditions in branches:
   493→            if condition is None:
   494→                selected: bool | None = True
   495→            else:
   496→                try:
   497→                    selected = self.eval_condition(condition, state, value, env)
   498→                except DispatcherBail:
   499→                    selected = None
   500→            if selected is False:
   501→                continue
   502→            branch_env = dict(env)
   503→            if leaf is None:
   504→                # `state = if c then v` with a false condition leaves the
   505→                # state unchanged; treat it as an edge to the same state.
   506→                leaf_value = value
   507→            else:
   508→                try:
   509→                    leaf_value = self.eval_state(leaf, state, value, branch_env)
   510→                except DispatcherBail:
   511→                    continue
   512→            branch_path = list(conditions) + (branch_conditions if selected is None else [])
   513→            result.extend(self.execute(statements[index + 1:], state, leaf_value, branch_path, list(actions), depth + 1, branch_env))
   514→        return result

   516→    def else_chain(self, statement: Node) -> list[Node]:
   517→        if not statement.get("elifs"):
   518→            return statement.get("else_", [])
   519→        condition, body = statement.get("elifs", [])[0]
   520→        nested = Node("if", condition.start, condition.end, cond=condition, then=body, elifs=statement.get("elifs", [])[1:], else_=statement.get("else_", []))
   521→        return [nested]

   523→    def fork_if(self, statement: Node, state: Binding, value: int | float, conditions: list[tuple[Node, bool]], actions: list[Node], depth: int, env: dict[int, Any] | None = None) -> list[DispatchPath]:
   524→        condition = statement.get("cond")
   525→        branch_env = dict(env or {})
   526→        result: list[DispatchPath] = []
   527→        result.extend(self.execute(statement.get("then", []), state, value, conditions + [(condition, True)], list(actions), depth + 1, branch_env))
   528→        current = list(statement.get("elifs", []))
   529→        prior = condition
   530→        for branch_condition, branch_body in current:
   531→            result.extend(self.execute(branch_body, state, value, conditions + [(prior, False), (branch_condition, True)], list(actions), depth + 1, branch_env))
   532→            prior = branch_condition
   533→        result.extend(self.execute(statement.get("else_", []), state, value, conditions + [(prior, False)], list(actions), depth + 1, branch_env))
   534→        return result

   536→    def contains_state(self, node: Node | None, state: Binding) -> bool:
   537→        if node is None:
   538→            return False
   539→        stack = [node]
   540→        while stack:
   541→            current = stack.pop()
   542→            if current.kind == "name" and binding_for(self.analyzer, current) is state:
   543→                return True
   544→            for value in current.fields.values():
   545→                if isinstance(value, Node):
   546→                    stack.append(value)
   547→                elif isinstance(value, (list, tuple)):
   548→                    stack.extend(item for item in value if isinstance(item, Node))
   549→                    for item in value:
   550→                        if isinstance(item, (list, tuple)):
   551→                            stack.extend(nested for nested in item if isinstance(nested, Node))
   552→        return False

   554→    def recover(self, dispatcher: Dispatcher) -> Node | None:
   555→        self.work = 0
   556→        self.emit_budget = 40000
   557→        graph: dict[Any, list[DispatchPath]] = {}
   558→        initial = self.apply_transform(dispatcher.initial, dispatcher.transform)
   559→        queue: list[tuple[int | float, dict[int, Any]]] = [(initial, dict(self.base_env))]
   560→        seen: dict[Any, dict[int, Any]] = {}
   561→        requeues: dict[Any, int] = {}
   562→        while queue:
   563→            value, incoming_env = queue.pop(0)
   564→            key = self.key(value)
   565→            if key in seen:
   566→                existing = seen[key]
   567→                if existing == incoming_env:
   568→                    continue
   569→                # Environment conflict: the state is reachable with
   570→                # different knowledge. Weaken to the shared subset and
   571→                # reprocess instead of giving up on the dispatcher.
   572→                merged = {k: v for k, v in incoming_env.items() if k in existing and existing[k] == v}
   573→                if merged == existing or requeues.get(key, 0) >= 3:
   574→                    continue
   575→                requeues[key] = requeues.get(key, 0) + 1
   576→                seen[key] = merged
   577→                incoming_env = merged
   578→            else:
   579→                if len(seen) >= self.max_states:
   580→                    raise DispatcherBail("state limit")
   581→                seen[key] = dict(incoming_env)
   582→            paths = self.execute(dispatcher.tree, dispatcher.state, value, [], [], env=incoming_env)
   583→            normalized: list[DispatchPath] = []
   584→            for path in paths:
   585→                if path.outcome[0] == "edge":
   586→                    next_value = self.apply_transform(path.outcome[1], dispatcher.transform)
   587→                    normalized.append(DispatchPath(path.conditions, path.actions, ("edge", next_value), path.env))
   588→                    next_key = self.key(next_value)
   589→                    if next_key not in seen or seen[next_key] != path.env:
   590→                        queue.append((next_value, path.env))
   591→                else:
   592→                    normalized.append(path)
   593→                if len(normalized) > self.max_paths:
   594→                    raise DispatcherBail("path limit")
   595→            graph[key] = normalized
   596→        self.stats.dispatchers_found += 1
   597→        self.stats.states_recovered += len(graph)
   598→        statements: list[Node] | None = None
   599→        try:
   600→            statements = self.emit_state(initial, graph, set(), 0)
   601→        except CycleHit:
   602→            for header in graph:
   603→                try:
   604→                    statements = self.emit_state(initial, graph, set(), 0, loop_header=header)
   605→                except CycleHit:
   606→                    continue
   607→                if statements is not None:
   608→                    break
   609→        if statements is None:
   610→            raise DispatcherBail("cyclic or ambiguous graph")
   611→        return Node("do", dispatcher.loop.start, dispatcher.loop.end, body=statements, states=len(graph))

   613→    def key(self, value: int | float) -> tuple[str, int | float]:
   614→        if isinstance(value, float) and value.is_integer():
   615→            value = int(value)
   616→        return (type(value).__name__, value)

   618→    def emit_state(self, value: int | float, graph: dict[Any, list[DispatchPath]], stack: set[Any], depth: int, loop_header: Any | None = None, in_loop: bool = False) -> list[Node] | None:
   619→        if depth > self.max_paths:
   620→            return None
   621→        key = self.key(value)
   622→        if key in stack:
   623→            if loop_header is not None and key == loop_header and in_loop:
   624→                return [Node("continue")]
   625→            raise CycleHit(key)
   626→        paths = graph.get(key)
   627→        if not paths:
   628→            return None
   629→        next_stack = set(stack)
   630→        next_stack.add(key)
   631→        if loop_header is not None and key == loop_header and not in_loop:
   632→            body = self.emit_paths(paths, graph, next_stack, depth + 1, True, loop_header)
   633→            if body is None:
   634→                return None
   635→            return [Node("while", cond=Node("bool", value=True), body=body)]
   636→        return self.emit_paths(paths, graph, next_stack, depth, in_loop, loop_header)

   638→    def emit_paths(self, paths: list[DispatchPath], graph: dict[Any, list[DispatchPath]], stack: set[Any], depth: int, in_loop: bool = False, loop_header: Any | None = None) -> list[Node] | None:
   639→        if not paths:
   640→            return []
   641→        self.emit_budget -= 1
   642→        if self.emit_budget < 0:
   643→            raise DispatcherBail("emit budget")
   644→        if len(paths) == 1 and not paths[0].conditions:
   645→            return self.emit_path(paths[0], graph, stack, depth, in_loop, loop_header)
   646→        unconditional = [path for path in paths if not path.conditions]
   647→        conditional = [path for path in paths if path.conditions]
   648→        if unconditional and conditional:
   649→            condition = conditional[0].conditions[0][0]
   650→            true_paths = [path for path in conditional if path.conditions[0][0] is condition and path.conditions[0][1]]
   651→            false_paths = [path for path in conditional if path.conditions[0][0] is condition and not path.conditions[0][1]]
   652→            if not true_paths and not false_paths:
   653→                return None
   654→            if not false_paths and len(unconditional) == 1:
   655→                false_paths = unconditional
   656→            if not true_paths or not false_paths or len(unconditional) > 1:
   657→                return None
   658→            true_paths = [DispatchPath(path.conditions[1:], path.actions, path.outcome, path.env) for path in true_paths]
   659→            false_paths = [DispatchPath(path.conditions[1:], path.actions, path.outcome, path.env) for path in false_paths]
   660→            then_body = self.emit_paths(true_paths, graph, set(stack), depth + 1, in_loop, loop_header)
   661→            else_body = self.emit_paths(false_paths, graph, set(stack), depth + 1, in_loop, loop_header)
   662→            if then_body is None or else_body is None:
   663→                return None
   664→            return [Node("if", condition.start, condition.end, cond=condition, then=then_body, elifs=[], else_=else_body)]
   665→        condition = paths[0].conditions[0][0] if paths[0].conditions else None
   666→        if condition is None:
   667→            return None
   668→        true_paths = [path for path in paths if path.conditions and path.conditions[0][0] is condition and path.conditions[0][1]]
   669→        false_paths = [path for path in paths if path.conditions and path.conditions[0][0] is condition and not path.conditions[0][1]]
   670→        if not true_paths or not false_paths:
   671→            return None
   672→        true_paths = [DispatchPath(path.conditions[1:], path.actions, path.outcome, path.env) for path in true_paths]
   673→        false_paths = [DispatchPath(path.conditions[1:], path.actions, path.outcome, path.env) for path in false_paths]
   674→        then_body = self.emit_paths(true_paths, graph, set(stack), depth + 1, in_loop, loop_header)
   675→        else_body = self.emit_paths(false_paths, graph, set(stack), depth + 1, in_loop, loop_header)
   676→        if then_body is None or else_body is None:
   677→            return None
   678→        return [Node("if", condition.start, condition.end, cond=condition, then=then_body, elifs=[], else_=else_body)]

   680→    def emit_path(self, path: DispatchPath, graph: dict[Any, list[DispatchPath]], stack: set[Any], depth: int, in_loop: bool = False, loop_header: Any | None = None) -> list[Node] | None:
   681→        self.emit_budget -= len(path.actions) + 2
   682→        if self.emit_budget < 0:
   683→            raise DispatcherBail("emit budget")
   684→        actions = [action.clone() for action in path.actions]
   685→        if actions and any(action.kind in {"local", "localfunc"} for action in actions):
   686→            actions = [Node("do", actions[0].start, actions[-1].end, body=actions)]
   687→        outcome = path.outcome
   688→        if outcome[0] == "break":
   689→            if in_loop:
   690→                actions.append(Node("break"))
   691→            return actions
   692→        if outcome[0] == "return":
   693→            actions.append(Node("return", actions[-1].end if actions else 0, values=[value.clone() for value in outcome[1]]))
   694→            return actions
   695→        if outcome[0] != "edge":
   696→            return None
   697→        successor = self.emit_state(outcome[1], graph, stack, depth + 1, loop_header, in_loop)
   698→        if successor is None:
   699→            return None
   700→        return actions + successor
