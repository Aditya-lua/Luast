    59→    59→    def apply(self, rounds: int = 8) -> int:
    60→    60→        completed = 0
    61→    61→        sys.setrecursionlimit(max(sys.getrecursionlimit(), 20000))
    62→    62→        for _ in range(rounds):
    63→    63→            changed = False
    64→    64→            lists = self.statement_lists(self.root)
    65→    65→            candidates: list[Dispatcher] = []
    66→    66→            for statements in lists:
    67→    67→                for index, statement in enumerate(statements):
    68→    68→                    if statement.kind != "while":
    69→    69→                        continue
    70→    70→                    candidate = self.recognize(statements, index, statement)
    71→    71→                    if candidate is not None:
    72→    72→                        candidates.append(candidate)
    73→    73→            candidates.sort(key=lambda item: item.loop.start, reverse=True)
    74→    74→            for candidate in candidates:
    75→    75→                try:
    76→    76→                    replacement = self.recover(candidate)
    77→    77→                except DispatcherBail:
    78→    78→                    continue
    79→    79→                except (RecursionError, OverflowError, ValueError, TypeError):
    80→    80→                    continue
    81→    81→                if replacement is None:
    82→    82→                    continue
    83→    83→                candidate.statements[candidate.index] = replacement
    84→    84→                self.stats.dispatchers_removed += 1
    85→    85→                completed += 1
    86→    86→                changed = True
    87→    87→            if not changed:
    88→    88→                break
    89→    89→        return completed

    91→    91→    def statement_lists(self, root: Node) -> list[list[Node]]:
    92→    92→        result: list[list[Node]] = []
    93→    93→        seen: set[int] = set()
    94→    94→
    95→    95→        def add(statements: object) -> None:
    96→    96→            if isinstance(statements, list) and id(statements) not in seen:
    97→    97→                seen.add(id(statements))
    98→    98→                result.append(statements)
    99→    99→
   100→   100→        add(root.fields.get("body", []))
   101→   101→        for node in walk(root):
   102→   102→            for key in ("body", "then", "else_"):
   103→   103→                add(node.fields.get(key))
   104→   104→            for _, body in node.fields.get("elifs", []):
   105→   105→                add(body)
   106→   106→        return result

   108→   108→    def recognize(self, statements: list[Node], index: int, loop: Node) -> Dispatcher | None:
   109→   109→        if not self.is_true(loop.get("cond")):
   110→   110→            return None
   111→   111→        body = loop.get("body", [])
   112→   112→        if not body:
   113→   113→            return None
   114→   114→        head = body[0]
   115→   115→        if head.kind != "assign" or len(head.get("targets", [])) != 1 or len(head.get("values", [])) != 1:
   116→   116→            return None
   117→   117→        target = head.get("targets", [])[0]
   118→   118→        if target.kind != "name":
   119→   119→            return None
   120→   120→        state = binding_for(self.analyzer, target)
   121→   121→        if state is None:
   122→   122→            return None
   123→   123→        transform = self.transition(head.get("values", [])[0], state)
   124→   124→        if transform is None:
   125→   125→            return None
   126→   126→        initial = self.initial_value(statements, index, state)
   127→   127→        if initial is None:
   128→   128→            return None
   129→   129→        tree = body[1:]
   130→   130→        if len(tree) == 1 and tree[0].kind == "do":
   131→   131→            tree = tree[0].get("body", [])
   132→   132→        if not self.looks_like_dispatch(tree, state):
   133→   133→            return None
   134→   134→        return Dispatcher(loop, statements, index, state, initial, transform, tree)

   136→   136→    def is_true(self, node: Node) -> bool:
   137→   137→        return isinstance(node, Node) and node.kind == "bool" and node.get("value") is True

   139→   139→    def transition(self, expression: Node, state: Binding) -> tuple[str, int | float] | None:
   140→   140→        if expression.kind == "binop" and expression.get("op") == "-":
   141→   141→            left = expression.get("left")
   142→   142→            right = expression.get("right")
   143→   143→            if self.is_state(right, state) and self.is_number_node(left):
   144→   144→                return ("complement", left.get("value"))
   145→   145→            if self.is_state(left, state) and self.is_number_node(right):
   146→   146→                return ("negate", right.get("value"))
   147→   147→        if expression.kind == "call":
   148→   148→            name = self.dotted_name(expression.get("func"))
   149→   149→            args = expression.get("args", [])
   150→   150→            if name == "bit32.bxor" and len(args) == 2 and self.is_state(args[0], state) and self.is_number_node(args[1]):
   151→   151→                return ("xor", args[1].get("value"))
   152→   152→            if name == "bit32.bxor" and len(args) == 2 and self.is_state(args[1], state) and self.is_number_node(args[0]):
   153→   153→                return ("xor", args[0].get("value"))
   154→   154→        return None

   156→   156→    def dotted_name(self, node: Node | None) -> str | None:
   157→   157→        if not isinstance(node, Node):
   158→   158→            return None
   159→   159→        if node.kind == "name":
   160→   160→            return node.get("name")
   161→   161→        if node.kind == "indexname":
   162→   162→            base = self.dotted_name(node.get("obj"))
   163→   163→            return base + "." + node.get("name", "") if base else None
   164→   164→        return None

   166→   166→    def is_state(self, node: Node | None, state: Binding) -> bool:
   167→   167→        return isinstance(node, Node) and node.kind == "name" and binding_for(self.analyzer, node) is state

   169→   169→    def is_number_node(self, node: Node | None) -> bool:
   170→   170→        return isinstance(node, Node) and node.kind == "number" and is_number(node.get("value"))

   172→   172→    def initial_value(self, statements: list[Node], index: int, state: Binding) -> int | float | None:
   173→   173→        for statement in reversed(statements[:index]):
   174→   174→            if statement.kind == "local":
   175→   175→                bindings = self.analyzer.declaration_bindings.get(id(statement), [])
   176→   176→                names = statement.get("names", [])
   177→   177→                for binding, value in zip(bindings, statement.get("values", [])):
   178→   178→                    if binding is state and self.is_number_node(value):
   179→   179→                        return value.get("value")
   180→   180→                if state.name in names:
   181→   181→                    return None
   182→   182→            if statement.kind == "assign":
   183→   183→                for target, value in zip(statement.get("targets", []), statement.get("values", [])):
   184→   184→                    if target.kind == "name" and binding_for(self.analyzer, target) is state:
   185→   185→                        return value.get("value") if self.is_number_node(value) else None
   186→   186→            if any(child.kind == "name" and binding_for(self.analyzer, child) is state for child in self.node_children(statement)):
   187→   187→                return None
   188→   188→        return None

   190→   190→    def node_children(self, node: Node) -> list[Node]:
   191→   191→        result: list[Node] = []
   192→   192→        for value in node.fields.values():
   193→   193→            if isinstance(value, Node):
   194→   194→                result.append(value)
   195→   195→            elif isinstance(value, (list, tuple)):
   196→   196→                for item in value:
   197→   197→                    if isinstance(item, Node):
   198→   198→                        result.append(item)
   199→   199→                    elif isinstance(item, (list, tuple)):
   200→   200→                        result.extend(item for item in item if isinstance(item, Node))
   201→   201→        return result

   203→   203→    def looks_like_dispatch(self, tree: list[Node], state: Binding) -> bool:
   204→   204→        found = False
   205→   205→        for node in tree:
   206→   206→            for child in self.node_children(node):
   207→   207→                if child.kind == "binop" and child.get("op") in {"<", "<=", ">", ">=", "==", "~="}:
   208→   208→                    if self.is_state(child.get("left"), state) or self.is_state(child.get("right"), state):
   209→   209→                        found = True
   210→   210→        if found:
   211→   211→            return True
   212→   212→        return any(node.kind in {"break", "continue", "return"} for node in tree)

   214→   214→    def apply_transform(self, value: int | float, transform: tuple[str, int | float]) -> int | float:
   215→   215→        kind, constant = transform
   216→   216→        if kind == "complement":
   217→   217→            return constant - value
   218→   218→        if kind == "negate":
   219→   219→            return value - constant
   220→   220→        return int(constant) ^ int(value)

   222→   222→    def eval_data(self, node: Node | None, state: Binding, value: int | float, env: dict[int, Any]) -> Any:
   223→   223→        if node is None:
   224→   224→            return UNKNOWN
   225→   225→        if self.is_state(node, state):
   226→   226→            return value
   227→   227→        if node.kind == "number" or node.kind == "string" or node.kind == "bool":
   228→   228→            return node.get("value")
   229→   229→        if node.kind == "nil":
   230→   230→            return None
   231→   231→        if node.kind == "name":
   232→   232→            binding = binding_for(self.analyzer, node)
   233→   233→            if binding is not None and binding.ident in env:
   234→   234→                return env[binding.ident]
   235→   235→            if binding is not None and binding.ident in self.base_env:
   236→   236→                return self.base_env[binding.ident]
   237→   237→            return UNKNOWN
   238→   238→        if node.kind == "paren":
   239→   239→            return self.eval_data(node.get("expr"), state, value, env)
   240→   240→        if node.kind == "unop":
   241→   241→            inner = self.eval_data(node.get("expr"), state, value, env)
   242→   242→            if inner is UNKNOWN:
   243→   243→                resolved = bool_const(node)
   244→   244→                return resolved if resolved is not None else UNKNOWN
   245→   245→            if node.get("op") == "not":
   246→   246→                return not truthy(inner)
   247→   247→            if node.get("op") == "-":
   248→   248→                return -inner
   249→   249→            if node.get("op") == "#" and isinstance(inner, bytes):
   250→   250→                return len(inner)
   251→   251→            return UNKNOWN
   252→   252→        if node.kind == "binop":
   253→   253→            operator = node.get("op")
   254→   254→            if operator == "and":
   255→   255→                left = self.eval_data(node.get("left"), state, value, env)
   256→   256→                if left is UNKNOWN:
   257→   257→                    resolved = bool_const(node)
   258→   258→                    return resolved if resolved is not None else UNKNOWN
   259→   259→                return self.eval_data(node.get("right"), state, value, env) if truthy(left) else left
   260→   260→            if operator == "or":
   261→   261→                left = self.eval_data(node.get("left"), state, value, env)
   262→   262→                if left is UNKNOWN:
   263→   263→                    resolved = bool_const(node)
   264→   264→                    return resolved if resolved is not None else UNKNOWN
   265→   265→                return left if truthy(left) else self.eval_data(node.get("right"), state, value, env)
   266→   266→            left = self.eval_data(node.get("left"), state, value, env)
   267→   267→            right = self.eval_data(node.get("right"), state, value, env)
   268→   268→            if left is UNKNOWN or right is UNKNOWN:
   269→   269→                return UNKNOWN
   270→   270→            if operator == "..":
   271→   271→                return left + right if isinstance(left, bytes) and isinstance(right, bytes) else UNKNOWN
   272→   272→            if operator in {"==", "~=", "<", "<=", ">", ">="}:
   273→   273→                try:
   274→   274→                    if operator == "==":
   275→   275→                        return left == right
   276→   276→                    if operator == "~=":
   277→   277→                        return left != right
   278→   278→                    if operator == "<":
   279→   279→                        return left < right
   280→   280→                    if operator == "<=":
   281→   281→                        return left <= right
   282→   282→                    if operator == ">":
   283→   283→                        return left > right
   284→   284→                    return left >= right
   285→   285→                except (TypeError, ValueError):
   286→   286→                    return UNKNOWN
   287→   287→            return numeric_result(operator, left, right)
   288→   288→        if node.kind == "call":
   289→   289→            name = self.dotted_name(node.get("func"))
   290→   290→            args = [self.eval_data(argument, state, value, env) for argument in node.get("args", [])]
   291→   291→            return builtin_call(name, args)
   292→   292→        if node.kind == "ifexpr":
   293→   293→            condition = self.eval_data(node.get("cond"), state, value, env)
   294→   294→            if condition is UNKNOWN:
   295→   295→                return UNKNOWN
   296→   296→            if truthy(condition):
   297→   297→                return self.eval_data(node.get("then"), state, value, env)
   298→   298→            for branch_condition, branch_value in node.get("elifs", []):
   299→   299→                branch = self.eval_data(branch_condition, state, value, env)
   300→   300→                if branch is UNKNOWN:
   301→   301→                    return UNKNOWN
   302→   302→                if truthy(branch):
   303→   303→                    return self.eval_data(branch_value, state, value, env)
   304→   304→            return self.eval_data(node.get("else_"), state, value, env)
   305→   305→        return UNKNOWN

   308→   308→        result = self.eval_data(node, state, value, env)
   309→   309→        if result is UNKNOWN or not is_number(result):
   310→   310→            raise DispatcherBail("dynamic state value")
   311→   311→        return result

   315→   315→        if result is UNKNOWN:

   319→   319→    def substitute_state(self, node: Node, state: Binding, value: int | float) -> Node:
   320→   320→        """Return a clone of node with every reference to the dispatcher
   321→   321→        state replaced by its known numeric value at this program point."""
   322→   322→        replacement = value_node(value)
   323→   323→
   324→   324→        def visit_pair(original: Node | None, clone: Node | None) -> None:
   325→   325→            if original is None or clone is None:
   326→   326→                return
   327→   327→            if original.kind == "function":
   328→   328→                raise StateSubstitutionError("state captured in payload closure")
   329→   329→            if original.kind == "name" and binding_for(self.analyzer, original) is state:
   330→   330→                clone.kind = replacement.kind
   331→   331→                clone.fields = dict(replacement.fields)
   332→   332→                return
   333→   333→            for key, field in original.fields.items():
   334→   334→                clone_field = clone.fields.get(key)
   335→   335→                if isinstance(field, Node) and isinstance(clone_field, Node):
   336→   336→                    visit_pair(field, clone_field)
   337→   337→                elif isinstance(field, (list, tuple)) and isinstance(clone_field, (list, tuple)):
   338→   338→                    visit_list(field, clone_field)
   339→   339→
   340→   340→        def visit_list(originals: Any, clones: Any) -> None:
   341→   341→            for original, clone in zip(originals, clones):
   342→   342→                if isinstance(original, Node) and isinstance(clone, Node):
   343→   343→                    visit_pair(original, clone)
   344→   344→                elif isinstance(original, (list, tuple)) and isinstance(clone, (list, tuple)):
   345→   345→                    visit_list(original, clone)
   346→   346→
   347→   347→        clone = node.clone()
   348→   348→        visit_pair(node, clone)
   349→   349→        return clone

   351→   351→    def execute(self, statements: list[Node], state: Binding, value: int | float, conditions: list[tuple[Node, bool]], actions: list[Node], depth: int = 0, env: dict[int, Any] | None = None) -> list[DispatchPath]:
   352→   352→        self.work += 1
   353→   353→        if self.work > self.max_work:
   354→   354→            raise DispatcherBail("work limit")
   355→   355→        if depth > 100:
   356→   356→            raise DispatcherBail("dispatcher nesting limit")
   357→   357→        paths: list[DispatchPath] = []
   358→   358→        current_value = value
   359→   359→        current_conditions = list(conditions)
   360→   360→        current_actions = list(actions)
   361→   361→        current_env = dict(env or {})
   362→   362→        for statement_index, statement in enumerate(statements):
   363→   363→            kind = statement.kind
   364→   364→            if kind == "assign":
   365→   365→                targets = statement.get("targets", [])
   366→   366→                values = statement.get("values", [])
   367→   367→                state_targets = [target for target in targets if target.kind == "name" and binding_for(self.analyzer, target) is state]
   368→   368→                if state_targets:
   369→   369→                    if len(targets) != 1 or len(values) != 1:
   370→   370→                        raise DispatcherBail("state assignment mixed with other targets")
   371→   371→                    try:
   372→   372→                        current_value = self.eval_state(values[0], state, current_value, current_env)
   373→   373→                    except DispatcherBail:
   374→   374→                        if values[0].kind == "ifexpr":
   375→   375→                            return self.expand_state_ifexpr(values[0], statements, statement_index, state, current_value, current_conditions, current_actions, current_env, depth)
   376→   376→                        raise
   377→   377→                    continue
   378→   378→                if self.contains_state(statement, state):
   379→   379→                    try:
   380→   380→                        current_actions.append(self.substitute_state(statement, state, current_value))
   381→   381→                    except StateSubstitutionError as exc:
   382→   382→                        raise DispatcherBail(str(exc))
   383→   383→                    if len(targets) == 1 and len(values) == 1 and targets[0].kind == "name":
   384→   384→                        target_binding = binding_for(self.analyzer, targets[0])
   385→   385→                        if target_binding is not None:
   386→   386→                            assigned = self.eval_data(values[0], state, current_value, current_env)
   387→   387→                            if assigned is UNKNOWN:
   388→   388→                                current_env.pop(target_binding.ident, None)
   389→   389→                            else:
   390→   390→                                current_env[target_binding.ident] = assigned
   391→   391→                    continue
   392→   392→                if len(targets) == 1 and len(values) == 1 and targets[0].kind == "name":
   393→   393→                    target_binding = binding_for(self.analyzer, targets[0])
   394→   394→                    if target_binding is not None:
   395→   395→                        assigned = self.eval_data(values[0], state, current_value, current_env)
   396→   396→                        if assigned is UNKNOWN:
   397→   397→                            current_env.pop(target_binding.ident, None)
   398→   398→                        else:
   399→   399→                            current_env[target_binding.ident] = assigned
   400→   400→                current_actions.append(statement)
   401→   401→                continue
   402→   402→            if kind == "local":
   403→   403→                if any(name in statement.get("names", []) for name in [state.name]) or self.contains_state(statement, state):
   404→   404→                    if self.contains_state(statement, state):
   405→   405→                        try:
   406→   406→                            current_actions.append(self.substitute_state(statement, state, current_value))
   407→   407→                        except StateSubstitutionError as exc:
   408→   408→                            raise DispatcherBail(str(exc))
   409→   409→                        for local_binding, local_value in zip(self.analyzer.declaration_bindings.get(id(statement), []), statement.get("values", [])):
   410→   410→                            assigned = self.eval_data(local_value, state, current_value, current_env)
   411→   411→                            if assigned is UNKNOWN:
   412→   412→                                current_env.pop(local_binding.ident, None)
   413→   413→                            else:
   414→   414→                                current_env[local_binding.ident] = assigned
   415→   415→                        continue
   416→   416→                    raise DispatcherBail("state in local declaration")
   417→   417→                for local_binding, local_value in zip(self.analyzer.declaration_bindings.get(id(statement), []), statement.get("values", [])):
   418→   418→                    assigned = self.eval_data(local_value, state, current_value, current_env)
   419→   419→                    if assigned is UNKNOWN:
   420→   420→                        current_env.pop(local_binding.ident, None)
   421→   421→                    else:
   422→   422→                        current_env[local_binding.ident] = assigned
   423→   423→                current_actions.append(statement)
   424→   424→                continue
   425→   425→            if kind == "if":
   426→   426→                try:
   427→   427→                    condition_value = self.eval_condition(statement.get("cond"), state, current_value, current_env)
   428→   428→                except DispatcherBail:
   429→   429→                    if self.contains_state(statement.get("cond"), state):
   430→   430→                        raise

   432→   432→                if condition_value:

   504→   504→                # `state = if c then v` with a false condition leaves the

   510→   510→                except DispatcherBail:

   536→   536→    def contains_state(self, node: Node | None, state: Binding) -> bool:
   537→   537→        if node is None:
   538→   538→            return False
   539→   539→        stack = [node]
   540→   540→        while stack:
   541→   541→            current = stack.pop()
   542→   542→            if current.kind == "name" and binding_for(self.analyzer, current) is state:
   543→   543→                return True
   544→   544→            for value in current.fields.values():
   545→   545→                if isinstance(value, Node):
   546→   546→                    stack.append(value)
   547→   547→                elif isinstance(value, (list, tuple)):
   548→   548→                    stack.extend(item for item in value if isinstance(item, Node))
   549→   549→                    for item in value:
   550→   550→                        if isinstance(item, (list, tuple)):
   551→   551→                            stack.extend(nested for nested in item if isinstance(nested, Node))
   552→   552→        return False

   554→   554→    def recover(self, dispatcher: Dispatcher) -> Node | None:
   555→   555→        self.work = 0
   556→   556→        self.emit_budget = 40000
   557→   557→        graph: dict[Any, list[DispatchPath]] = {}
   558→   558→        initial = self.apply_transform(dispatcher.initial, dispatcher.transform)
   559→   559→        queue: list[tuple[int | float, dict[int, Any]]] = [(initial, dict(self.base_env))]
   560→   560→        seen: dict[Any, dict[int, Any]] = {}
   561→   561→        requeues: dict[Any, int] = {}
   562→   562→        while queue:
   563→   563→            value, incoming_env = queue.pop(0)
   564→   564→            key = self.key(value)
   565→   565→            if key in seen:
   566→   566→                existing = seen[key]
   567→   567→                if existing == incoming_env:
   568→   568→                    continue
   569→   569→                # Environment conflict: the state is reachable with
   570→   570→                # different knowledge. Weaken to the shared subset and
   571→   571→                # reprocess instead of giving up on the dispatcher.
   572→   572→                merged = {k: v for k, v in incoming_env.items() if k in existing and existing[k] == v}
   573→   573→                if merged == existing or requeues.get(key, 0) >= 3:
   574→   574→                    continue
   575→   575→                requeues[key] = requeues.get(key, 0) + 1
   576→   576→                seen[key] = merged
   577→   577→                incoming_env = merged
   578→   578→            else:
   579→   579→                if len(seen) >= self.max_states:
   580→   580→                    raise DispatcherBail("state limit")
   581→   581→                seen[key] = dict(incoming_env)
   582→   582→            paths = self.execute(dispatcher.tree, dispatcher.state, value, [], [], env=incoming_env)
   583→   583→            normalized: list[DispatchPath] = []
   584→   584→            for path in paths:
   585→   585→                if path.outcome[0] == "edge":
   586→   586→                    next_value = self.apply_transform(path.outcome[1], dispatcher.transform)
   587→   587→                    normalized.append(DispatchPath(path.conditions, path.actions, ("edge", next_value), path.env))
   588→   588→                    next_key = self.key(next_value)
   589→   589→                    if next_key not in seen or seen[next_key] != path.env:
   590→   590→                        queue.append((next_value, path.env))
   591→   591→                else:
   592→   592→                    normalized.append(path)
   593→   593→                if len(normalized) > self.max_paths:
   594→   594→                    raise DispatcherBail("path limit")
   595→   595→            graph[key] = normalized
   596→   596→        self.stats.dispatchers_found += 1
   597→   597→        self.stats.states_recovered += len(graph)
   598→   598→        statements: list[Node] | None = None
   599→   599→        try:
   600→   600→            statements = self.emit_state(initial, graph, set(), 0)
   601→   601→        except CycleHit:
   602→   602→            for header in graph:
   603→   603→                try:
   604→   604→                    statements = self.emit_state(initial, graph, set(), 0, loop_header=header)
   605→   605→                except CycleHit:
   606→   606→                    continue
   607→   607→                if statements is not None:
   608→   608→                    break
   609→   609→        if statements is None:
   610→   610→            raise DispatcherBail("cyclic or ambiguous graph")
   611→   611→        return Node("do", dispatcher.loop.start, dispatcher.loop.end, body=statements, states=len(graph))

   680→   680→    def emit_path(self, path: DispatchPath, graph: dict[Any, list[DispatchPath]], stack: set[Any], depth: int, in_loop: bool = False, loop_header: Any | None = None) -> list[Node] | None:

   682→   682→        if self.emit_budget < 0:
   683→   683→            raise DispatcherBail("emit budget")
   684→   684→        actions = [action.clone() for action in path.actions]
   685→   685→        if actions and any(action.kind in {"local", "localfunc"} for action in actions):
   686→   686→            actions = [Node("do", actions[0].start, actions[-1].end, body=actions)]

   689→   689→            if in_loop:
   690→   690→                actions.append(Node("break"))

   693→   693→            actions.append(Node("return", actions[-1].end if actions else 0, values=[value.clone() for value in outcome[1]]))

   696→   696→            return None

   698→   698→        if successor is None:
   699→   699→            return None
   700→   700→        return actions + successor
